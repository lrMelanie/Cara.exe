import hashlib
def calculate_checksum(file_path):
try:
sha = hashlib.sha256()
with open(file_path, 'rb') as f:
while chunk := f.read(4096):
sha.update(chunk)
return sha.hexdigest()
except IOError:
print(f"IO_ERROR:0x08X")
def verify_file(original_hash, file_path):
current_hash = calculate_checksum(file_path)
if current_hash == original_hash:
print("INTEGRITY_CHECK_PASSED")
return True
else:
print(f"TAMPER_DETECTED")
return False
def simple_encrypt(text, shift=3):
result = ""
for char in text:
result += chr(ord(char) + shift)
return result
