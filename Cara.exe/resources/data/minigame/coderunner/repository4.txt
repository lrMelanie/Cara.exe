from model import translate, classify, generate
import speech_recognition as sr
SECURITY_CODES = ["SL-000", "TERMINATE", "QUARANTINE"]
recognizer = sr.Recognizer()
system_active = False
def auth_check(text: str) -> bool:
return any(code in text.upper() for code in SECURITY_CODES)
def activation_listener() -> None:
global system_active
with sr.Microphone(sample_rate=44100) as mic:
while True:
try: audio = recognizer.listen(mic, timeout=1.5)
text = recognizer.recognize_google(audio)
if auth_check(text):
system_active = not system_active
speak_text(f"System {'ACTIVATED' if system_active else 'OFFLINE'}")
if system_active: command_handler()
except sr.UnknownValueError: pass
def command_handler() -> None:
while system_active:
try: with sr.Microphone() as mic:
audio = recognizer.listen(mic, phrase_time_limit=4)
command = recognizer.recognize_google(audio)
if "disable" in command.lower():
system_active = False
speak_text("Shutdown sequence initiated")
translated = translate(command)
category = classify(translated)
response = generate(translated)
speak_text(f"[{category}] {response}")
except Exception as e:
logging.error(f"CMD_ERROR:0x{hash(e):X}")
